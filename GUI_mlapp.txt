    % Callbacks that handle component events
    methods (Access = private)

        % Button pushed function: RunButton
        function RunButtonPushed(app, event)
            % User inputs
            R1 = app.R1DropDown.Value;
            R2 = app.R2DropDown.Value;
            T1 = app.T1CEditField.Value;
            T2 = app.T2CEditField.Value;
            N1 = app.N1DropDown.Value;
            N2 = app.N2DropDown.Value;
            Nb = app.NbDropDown.Value;
            I1 = app.I1nominalDropDown.Value;
            A20_gain = app.A20GainDropDown.Value;
            S_ab = app.SabEditField.Value;
            DeltaT_ab = app.DeltaTabEditField.Value;
            FlagSpecificTau = app.SwitchSpecificTau.Value;
            SpecificTau = app.SpecificTauEditField.Value;
            
            % Sends the parameter values ​​to the program
            in_bal_estimator_v3(R1,R2,T1,T2,N1,N2,Nb,I1,A20_gain,S_ab,DeltaT_ab,FlagSpecificTau,SpecificTau);

            % Receives the arrays from the program to plot the graph and the calculated noise (Allan deviation) of the balance current for an specific tau
            [tauArray, In_balArray,AD4SpecificTau] = in_bal_estimator_v3(R1,R2,T1,T2,N1,N2,Nb,I1,A20_gain,S_ab,DeltaT_ab,FlagSpecificTau,SpecificTau);
            
            % Output :: sigma-tau graph (customized configuration)
            plot(app.UIAxes, tauArray, In_balArray, 'Color', [0.4 0 0.4], 'LineWidth', 2);
            app.UIAxes.XScale = 'log';
            app.UIAxes.YScale = 'log';
            xlabel(app.UIAxes, 'Intervalo de medição \tau (s)');
            ylabel(app.UIAxes, 'Desvio de Allan (nA)');
            app.UIAxes.XLim = [10 100];
            if app.SwitchYScale.Value == "Fixa"
                app.UIAxes.YLim = [1 100];
            else
                app.UIAxes.YLimMode = 'auto';
            end
            grid(app.UIAxes, 'on');
                  
            % Output :: noise (Allan deviation) of the balance current for an specific tau
            app.ResultSpecificTauLabel.Text = [num2str(AD4SpecificTau)];
        end

        % Button pushed function: ClearButton
        function ClearButtonPushed(app, event)
            % Reset the graph and the result label that shows the estimated balance current noise (in terms of Allan deviation) for an specific tau
            cla(app.UIAxes);
            app.ResultSpecificTauLabel.Text = "";
            % app.SpecificTauEditField.Value = 10;
        end

        % Value changed function: SwitchYScale
        function YScaleMode(app, event)
            % Toggles the scale mode of the y-axis between fixed and automatic - even after code execution
            if app.SwitchYScale.Value == "Fixa"
                app.UIAxes.YLim = [1 100];
            else
                app.UIAxes.YLimMode = 'auto';
            end
        end
    end
